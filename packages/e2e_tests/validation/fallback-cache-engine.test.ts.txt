it("fallback requires a new complete snapshot for each scope and isolates the newer database", async () => {
  const upgraded = await openDB("daynight-task-cache", 14, { upgrade(database) { database.createObjectStore("keep"); } });
  await upgraded.put("keep", "untouched", "sentinel");
  const first = scope(); await seed(first, [task(1)]); first.dispose();
  const anotherAccount = scope(otherUser); await seed(anotherAccount, [task(3)]); anotherAccount.dispose();
  const anotherProject = scope(); await seed(anotherProject, [task(4)], { ...project, id: otherProject }); anotherProject.dispose();
  const database = await inspect();
  const unrelated = [[otherUser, projectId, taskId(3)], [userId, otherProject, taskId(4)]] as const;
  for (const [owner, projectKey, item] of unrelated) {
    await database.put("details", { userId: owner, projectId: projectKey, taskId: item, sentinel: "detail" } as never);
    await database.put("sessionHistory", { userId: owner, projectId: projectKey, taskId: item, sentinel: "history" } as never);
    await database.put("profiles", { userId: owner, projectId: projectKey, profileUserId: item, sentinel: "profile" } as never);
  }
  const otherRows = async () => await Promise.all(unrelated.map(async ([owner, projectKey]) => ({
    owner: await database.get("userScopes", owner), project: await database.get("projects", [owner, projectKey]),
    summaries: await database.getAllFromIndex("summaries", "by-project", [owner, projectKey]),
    details: await database.getAllFromIndex("details", "by-project", [owner, projectKey]),
    histories: await database.getAllFromIndex("sessionHistory", "by-project", [owner, projectKey]),
    profiles: await database.getAllFromIndex("profiles", "by-project", [owner, projectKey]),
  })));
  const before = await otherRows();
  expect(before.every(rows => rows.project && rows.summaries.length === 1 && rows.details.length === 1 && rows.histories.length === 1 && rows.profiles.length === 1)).toBe(true);
  const reopened = scope();
  const ticket = await reopened.beginSync(projectId, signal());
  expect(ticket.epoch).toBeNull(); expect(ticket.maxUpdatedAt).toBeNull();
  expect(reopened.canPreviewProject(projectId)).toBe(false);
  expect(await reopened.readProject(projectId, signal())).toBeNull();
  await expect(reopened.detailTransaction(ticket, signal(), async () => true)).rejects.toMatchObject({ name: "AbortError" });
  await expect(reopened.sessionHistoryTransaction(ticket, signal(), async () => true)).rejects.toMatchObject({ name: "AbortError" });
  await expect(reopened.applySync(ticket, response([task(1)]), signal())).rejects.toMatchObject({ kind: "protocol" });
  const result = await reopened.applySync(ticket, response([task(2)], { mode: "snapshot", epoch: nextEpoch }), signal());
  expect(result.project.manifest.epoch).toBe(nextEpoch);
  expect([...result.project.summaries.keys()]).toEqual([taskId(2)]);
  expect(reopened.canPreviewProject(projectId)).toBe(true);
  expect(await otherRows()).toEqual(before);
  expect(await upgraded.get("keep", "sentinel")).toBe("untouched");
  upgraded.close();
});

it("fallback withdraws a live admission on failure and requires another full snapshot", async () => {
  const live = scope(); await seed(live, [task(1)]);
  const before = await live.beginSync(projectId, signal());
  const direct = await live.beginTaskRead(projectId, taskId(1), 1, 1, signal());
  const events: unknown[] = []; live.subscribe(event => events.push(event));
  live.withdrawRecoveryProject(projectId, new Error("offline"));
  expect(live.getAuthorizedProject(projectId)).toBeNull();
  expect(live.canPreviewProject(projectId)).toBe(false);
  expect(live.isTaskReadCurrent(direct)).toBe(false);
  expect(events).toContainEqual(expect.objectContaining({ accessRevoked: true, directAccessUncertain: true }));
  await expect(live.detailTransaction(before, signal(), async () => true)).rejects.toMatchObject({ name: "AbortError" });
  const fresh = await live.beginSync(projectId, signal()); expect(fresh.epoch).toBeNull();
  await live.applySync(fresh, response([task(2)], { mode: "snapshot", epoch: nextEpoch }), signal());
  await vi.waitFor(() => expect(events).toContainEqual(expect.objectContaining({ accessRestored: true, directAccessRecheck: true })));
  expect(live.getAuthorizedProject(projectId)?.summaries.has(taskId(1))).toBe(false);
});
