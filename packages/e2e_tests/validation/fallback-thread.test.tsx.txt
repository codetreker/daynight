const id = (value: number) => `00000000-0000-4000-8000-${String(value).padStart(12, '0')}`;
const projectId = id(100), taskId = id(200);
const message = (value: number, body: string, extra: Partial<ThreadMessage> = {}): ThreadMessage => ({ id: id(value), author: { userId: TEST_VIEWER_ID, displayName: 'Alice', Agent: false }, body, createdAt: `2026-11-03T08:${String(value).padStart(2, '0')}:00.123Z`, ...extra });
const resource = (value: unknown, status = 200) => new Response(JSON.stringify(value), { status, headers: viewerHeaders(TEST_VIEWER_ID, { 'Content-Type': 'application/json' }) });
const page = (items: readonly ThreadMessage[], hasMoreBefore = false, hasMoreAfter = false) => resource({ items, hasMoreBefore, hasMoreAfter });
const Harness = ({ active = true, occupied = () => {}, admissionBlocked = false, parentUnavailable = false, preserveHistoryDraft = false, historyAdmitted = true, historyRevision = 0, trackDraft = false }: { readonly active?: boolean; readonly occupied?: (value: boolean) => void; readonly admissionBlocked?: boolean; readonly parentUnavailable?: boolean; readonly preserveHistoryDraft?: boolean; readonly historyAdmitted?: boolean; readonly historyRevision?: number; readonly trackDraft?: boolean }) => {
  const owner = useRef<HTMLDivElement>(null);
  return <CurrentAccountProvider user={{ id: TEST_VIEWER_ID, name: 'Current Alice', avatarUrl: null }} admissionBlocked={admissionBlocked}><div ref={owner}><TaskDiscussion projectId={projectId} taskId={taskId} active={active} owner={owner as RefObject<HTMLElement | null>} onDetailsOpenChange={occupied} parentUnavailable={parentUnavailable} preserveHistoryDraft={preserveHistoryDraft} historyAdmitted={historyAdmitted} historyRevision={historyRevision} trackDraft={trackDraft} /></div></CurrentAccountProvider>;
};
const render = (ui = <Harness />) => renderWithLocale(ui, { locale: 'zh-CN', taskCache: true });


it.each(['denied', 'offline'])('fallback withdraws live discussion and project admission on %s', async (failure) => {
  const scope = createTaskCacheScope({ userId: TEST_VIEWER_ID, onIdentityMismatch: vi.fn() });
  const observed: unknown[] = []; scope.subscribe(event => observed.push(event));
  const fetcher = vi.fn()
    .mockResolvedValueOnce(page([message(1, 'Previously visible history')]))
    .mockImplementationOnce(async () => {
      if (failure === 'offline') throw new TypeError('Network unavailable');
      return resource({ error: { code: 'forbidden', message: 'forbidden', requestId: 'refused' } }, 403);
    });
  vi.stubGlobal('fetch', fetcher);
  try {
    const view = renderWithLocale(<Harness />, { locale: 'zh-CN', taskCache: { scope } });
    await screen.findByText('Previously visible history');
    fireEvent.click(view.container.querySelector('.thread-bar button')!);
    await waitFor(() => expect(observed).toContainEqual(expect.objectContaining({ projectId, accessRevoked: true, directAccessUncertain: true })));
    expect(screen.queryByText('Previously visible history')).toBeNull();
    expect(fetcher).toHaveBeenCalledTimes(2);
    view.unmount();
  } finally { scope.dispose(); }
});

it('withdraws after a restored history invalidates a successful reference lookup', async () => {
  const { createThreadController } = await import('../src/features/task-thread/controller');
  const { ApiError } = await import('../src/api/client');
  const cache = createTaskCacheScope({ userId: TEST_VIEWER_ID, onIdentityMismatch: vi.fn() });
  const events: unknown[] = []; let controller: ReturnType<typeof createThreadController>;
  cache.subscribe(event => { events.push(event); if (event.accessRevoked) controller.retire(); });
  const before = Promise.withResolvers<import('../src/features/task-thread/contract').ThreadPage>();
  const after = Promise.withResolvers<import('../src/features/task-thread/contract').ThreadPage>();
  const list = vi.fn().mockResolvedValueOnce({ items: [message(10, 'Prior history')], hasMoreBefore: false, hasMoreAfter: false })
    .mockReturnValueOnce(before.promise).mockReturnValueOnce(after.promise);
  controller = createThreadController({ scope: { userId: TEST_VIEWER_ID, projectId, taskId }, lifetime: cache.signal,
    ready: async () => {}, principalMismatch: cache.principalMismatch, onReadFailure: error => cache.withdrawRecoveryProject(projectId, error),
    api: { list, post: vi.fn() } });
  try {
    controller.setActive(true); await controller.refresh();
    expect(controller.snapshot().visible).toHaveLength(1);
    const reference = controller.showReference(id(20));
    await waitFor(() => expect(list).toHaveBeenCalledTimes(2));
    expect(list.mock.calls[1]![1]).toEqual({ beforeId: id(20), count: 1 });
    before.resolve({ items: [], hasMoreBefore: false, hasMoreAfter: false });
    await waitFor(() => expect(list).toHaveBeenCalledTimes(3));
    expect(list.mock.calls[2]![1]).toEqual({ fromBeginning: true, count: 1 });
    // Restored history returns a different first message despite both requests succeeding.
    after.resolve({ items: [message(30, 'Restored history')], hasMoreBefore: false, hasMoreAfter: false });
    await reference;
    expect(events).toContainEqual(expect.objectContaining({ accessRevoked: true, error: expect.any(ApiError) }));
    expect(controller.snapshot()).toMatchObject({ retired: true, visible: [] });
  } finally { controller.retire(); cache.dispose(); }
});

it.each(['failure', 'success', 'retired'])('does not withdraw for a stale reference %s', async mode => {
  const { createThreadController } = await import('../src/features/task-thread/controller');
  const { ApiError } = await import('../src/api/client');
  const cache = createTaskCacheScope({ userId: TEST_VIEWER_ID, onIdentityMismatch: vi.fn() });
  const events: unknown[] = []; cache.subscribe(event => events.push(event));
  const pending = Promise.withResolvers<import('../src/features/task-thread/contract').ThreadPage>();
  const list = vi.fn().mockResolvedValueOnce({ items: [message(10, 'Prior history')], hasMoreBefore: false, hasMoreAfter: false }).mockReturnValueOnce(pending.promise);
  const controller = createThreadController({ scope: { userId: TEST_VIEWER_ID, projectId, taskId }, lifetime: cache.signal,
    ready: async () => {}, principalMismatch: cache.principalMismatch, onReadFailure: error => cache.withdrawRecoveryProject(projectId, error),
    api: { list, post: vi.fn() } });
  try {
    controller.setActive(true); await controller.refresh();
    const reference = controller.showReference(id(20)); await waitFor(() => expect(list).toHaveBeenCalledTimes(2));
    if (mode === 'retired') controller.retire(); else controller.setActive(false);
    if (mode === 'failure') pending.reject(new ApiError('network'));
    else pending.resolve({ items: [message(30, 'Late history')], hasMoreBefore: false, hasMoreAfter: false });
    await reference;
    expect(events).not.toContainEqual(expect.objectContaining({ accessRevoked: true }));
    expect(controller.snapshot().visible.some(value => value.body === 'Late history')).toBe(false);
  } finally { controller.retire(); cache.dispose(); }
});
