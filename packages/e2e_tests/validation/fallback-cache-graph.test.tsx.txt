it('fallback graph retries when a concurrent direct snapshot commits first', async () => {
  const cache = scope(), pending = deferred<TaskSync>(), current = graph();
  const snapshot = taskSyncFromGraph(current, { versionForTask: () => 1n });
  vi.mocked(getTaskSync)
    .mockImplementationOnce(async () => taskSyncReadFixture(await pending.promise))
    .mockImplementationOnce(async () => taskSyncReadFixture(snapshot))
    .mockImplementationOnce(async () => taskSyncReadFixture(sync(current)));
  const view = mount(cache);
  await waitFor(() => expect(getTaskSync).toHaveBeenCalledTimes(1));
  const fence = await cache.beginTaskRead(projectId, taskId, 1, 1, new AbortController().signal);
  expect(cache.isTaskReadCurrent(fence)).toBe(true);
  await act(async () => { pending.resolve(snapshot); });
  await waitFor(() => expect(getTaskSync).toHaveBeenCalledTimes(3));
  await waitFor(() => expect(view.result.current.snapshot?.index.nodes.has(taskId)).toBe(true));
  expect(view.result.current.error).toBeNull();
});
