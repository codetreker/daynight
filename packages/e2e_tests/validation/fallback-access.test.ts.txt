it.each([new ApiError("server", 403, "forbidden"), new ApiError("network")])("fallback withdraws project data after a live access read fails", async (failure) => {
  const cache = scope(); await seed(cache, [task(1)]);
  const events: unknown[] = []; cache.subscribe(event => events.push(event));
  const access = new ProjectAccessSession(projectId, cache, false); const stop = access.start(false);
  vi.mocked(getProjectAccess).mockResolvedValueOnce(directAccessFixture(projectId, userId)).mockRejectedValueOnce(failure);
  try {
    await access.revalidate(); expect(access.snapshot().status).toBe("ready");
    expect(cache.getAuthorizedProject(projectId)).not.toBeNull();
    await access.revalidate();
    expect(access.snapshot()).toMatchObject({ status: "error", access: null, error: failure });
    expect(cache.getAuthorizedProject(projectId)).toBeNull();
    expect(events).toContainEqual(expect.objectContaining({ accessRevoked: true, error: failure }));
  } finally { stop(); }
});
