// Production candidate processes exercise the deployed entry point. The controlled
// Webhook process is used only for the isolated transport/effect boundary.
func startProductionCandidate(t *testing.T, ctx context.Context, values map[string]string) *persistenceProcess {
	t.Helper()
	cmd := exec.CommandContext(ctx, os.Getenv("V1_RUNTIME")+"/bin/server-candidate", "--root", testStorageRoot(t))
	cmd.Env = grantEnvironment(values)
	logs := &persistenceProcessLog{}
	cmd.Stdout, cmd.Stderr = logs, logs
	if err := cmd.Start(); err != nil {
		t.Fatal(err)
	}
	p := &persistenceProcess{cmd: cmd, finished: make(chan error, 1), logs: logs}
	go func() { p.finished <- cmd.Wait() }()
	t.Cleanup(func() {
		if !p.stopped {
			if err := p.stop(); err != nil {
				t.Error(err)
			}
		}
	})
	return p
}

func TestV1PrecommitLockFailureRestoresOldServer(t *testing.T) {
	values := processAPIEnvironment(t)
	values["DAYNIGHT_WEBHOOK_ENABLED"] = "0"
	ctx, cancel := context.WithTimeout(t.Context(), 45*time.Second)
	defer cancel()
	startOld := func() *persistenceProcess {
		cmd := exec.CommandContext(ctx, os.Getenv("V1_RUNTIME")+"/bin/server122", "--root", testStorageRoot(t))
		cmd.Env = grantEnvironment(values)
		logs := &persistenceProcessLog{}
		cmd.Stdout, cmd.Stderr = logs, logs
		if err := cmd.Start(); err != nil {
			t.Fatal(err)
		}
		p := &persistenceProcess{cmd: cmd, finished: make(chan error, 1), logs: logs}
		go func() { p.finished <- cmd.Wait() }()
		t.Cleanup(func() {
			if !p.stopped {
				if err := p.stop(); err != nil {
					t.Error(err)
				}
			}
		})
		waitMigrationProcessReady(t, ctx, p, values["DAYNIGHT_PUBLIC_ORIGIN"])
		return p
	}
	old := startOld()
	if err := old.stop(); err != nil {
		t.Fatal(err)
	}
	locker, err := pgx.Connect(ctx, values["DAYNIGHT_DATABASE_URL"])
	if err != nil {
		t.Fatal(err)
	}
	defer locker.Close(context.Background())
	var ledger string
	if err := locker.QueryRow(ctx, `SELECT jsonb_agg(to_jsonb(m) ORDER BY version)::text FROM schema_migrations m HAVING max(version)=122`).Scan(&ledger); err != nil {
		t.Fatal(err)
	}
	tx, err := locker.Begin(ctx)
	if err != nil {
		t.Fatal(err)
	}
	defer tx.Rollback(context.Background())
	if _, err := tx.Exec(ctx, `LOCK TABLE tasks IN ACCESS EXCLUSIVE MODE`); err != nil {
		t.Fatal(err)
	}
	values["DAYNIGHT_STARTUP_TIMEOUT"] = "2s"
	failed := startProductionCandidate(t, ctx, values)
	select {
	case err := <-failed.finished:
		failed.stopped = true
		if err == nil || !strings.Contains(failed.logs.String(), "database migrate failed") {
			t.Fatalf("candidate did not fail migration under lock: %v %s", err, failed.logs.String())
		}
	case <-ctx.Done():
		t.Fatal("candidate did not stop under migration lock")
	}
	if err := tx.Rollback(ctx); err != nil {
		t.Fatal(err)
	}
	fresh, err := pgx.Connect(ctx, values["DAYNIGHT_DATABASE_URL"])
	if err != nil {
		t.Fatal(err)
	}
	defer fresh.Close(context.Background())
	var same bool
	if err := fresh.QueryRow(ctx, `SELECT max(version)=122 AND jsonb_agg(to_jsonb(m) ORDER BY version)::text=$1 FROM schema_migrations m`, ledger).Scan(&same); err != nil || !same {
		t.Fatalf("precommit ledger changed: %v %v", same, err)
	}
	delete(values, "DAYNIGHT_STARTUP_TIMEOUT")
	restored := startOld()
	if err := restored.stop(); err != nil {
		t.Fatal(err)
	}
	t.Log("candidate lock deadline before commit; fresh connection confirms identical baseline122 ledger; actual old server returned ready")
}
